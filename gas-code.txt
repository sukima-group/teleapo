// アポインターがログインした時に呼ばれる（架電リスト取得）
function doGet(e) {
  try {
    // ping 接続テスト (v15: ブラウザから直接アクセスしたとき用)
    if (!e || !e.parameter || (!e.parameter.pin && !e.parameter.action)) {
      return ContentService.createTextOutput(JSON.stringify({
        status: "ok",
        message: "GAS Web App is alive",
        version: "v60",
        timestamp: new Date().toISOString(),
        note: "Use ?pin=XXX to fetch list, or POST with type field"
      })).setMimeType(ContentService.MimeType.JSON);
    }
    if (e.parameter.action === "ping") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "ok",
        message: "pong",
        timestamp: new Date().toISOString()
      })).setMimeType(ContentService.MimeType.JSON);
    }
    var pin = e.parameter.pin;
    var sheetName = PIN_MAP[pin];

    if (!sheetName) {
      return ContentService.createTextOutput(
        JSON.stringify({error: "不明なPIN: " + pin})
      ).setMimeType(ContentService.MimeType.JSON);
    }

    var ss = SpreadsheetApp.openById(SHEET_ID);
    var sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      // v46: 新メンバーの初回ログイン - シートを自動作成し「リスト未登録」として正常応答 (エラーにしない)
      try { sheet = ss.insertSheet(sheetName); writeListHeader_(sheet); } catch (mkErr) {}
      return ContentService.createTextOutput(
        JSON.stringify({error: "データが空です", created: true, rows: [], cm: {}})
      ).setMimeType(ContentService.MimeType.JSON);
    }

    var data = sheet.getDataRange().getValues();
    if (data.length < 2) {
      return ContentService.createTextOutput(
        JSON.stringify({error: "データが空です", rows: [], cm: {}})
      ).setMimeType(ContentService.MimeType.JSON);
    }

    var hdr = data[0];
    var rows = [];
    for (var i = 1; i < data.length; i++) {
      var hasContent = false;
      for (var j = 0; j < data[i].length; j++) {
        if (data[i][j] !== "") { hasContent = true; break; }
      }
      if (hasContent) rows.push(data[i]);
    }

    var lh = hdr.map(function(h) { return String(h).toLowerCase(); });
    function findCol(patterns) {
      for (var i = 0; i < lh.length; i++) {
        for (var j = 0; j < patterns.length; j++) {
          if (lh[i].indexOf(patterns[j]) >= 0) return i;
        }
      }
      return -1;
    }

    var cm = {
      co: findCol(["施設", "会社", "法人", "名称", "名前"]),
      ph: findCol(["電話", "tel", "番号"]),
      pe: findCol(["担当", "氏名", "施設長"]),
      ad: findCol(["住所", "所在地"]),
      src: findCol(["元行"])   // v49: 元スプレッドシートの行番号（あれば確実に突合できる）
    };

    return ContentService.createTextOutput(
      JSON.stringify({rows: rows, cm: cm})
    ).setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(
      JSON.stringify({error: String(err)})
    ).setMimeType(ContentService.MimeType.JSON);
  }
}

// 勤怠 / 架電結果 / 再架電 / アポ詳細 / シフト申請 / 退勤レポート の書き込み
function doPost(e) {
  try {
    // 防御的パース: postData がない場合は parameter からも取得を試みる
    var raw = "";
    if (e && e.postData && e.postData.contents) {
      raw = e.postData.contents;
    } else if (e && e.parameter && e.parameter.payload) {
      raw = e.parameter.payload;
    }
    if (!raw) {
      return ContentService.createTextOutput("error: no payload")
        .setMimeType(ContentService.MimeType.TEXT);
    }
    var d;
    try {
      d = JSON.parse(raw);
    } catch (parseErr) {
      Logger.log("JSON parse error. raw: " + String(raw).slice(0, 200));
      return ContentService.createTextOutput("error: invalid JSON: " + parseErr.message)
        .setMimeType(ContentService.MimeType.TEXT);
    }
    if (!d || !d.type) {
      return ContentService.createTextOutput("error: missing type")
        .setMimeType(ContentService.MimeType.TEXT);
    }

    // 接続テスト用 (v15追加)
    if (d.type === "ping") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "ok",
        message: "GAS connected",
        timestamp: new Date().toISOString(),
        version: "v60"
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ===== v20: idempotency チェック (write系のみ) =====
    // fetch_state は読み取りなのでスキップ (毎回最新を返す必要)
    var READ_ONLY_TYPES = ["fetch_state", "fetch_live", "status_update"];  // v29: 現況系もidempotencyスキップ(常に最新を処理)
    if (d.clientReqId && READ_ONLY_TYPES.indexOf(d.type) < 0) {
      var idempCache = CacheService.getScriptCache();
      var idempKey = "req:" + d.clientReqId;
      var idempSeen = idempCache.get(idempKey);
      if (idempSeen) {
        Logger.log("Duplicate request detected: " + d.clientReqId + " (type=" + d.type + ")");
        return ContentService.createTextOutput(JSON.stringify({
          status: "duplicate",
          type: d.type,
          message: "already processed",
          firstSeenAt: idempSeen
        })).setMimeType(ContentService.MimeType.JSON);
      }
      idempCache.put(idempKey, new Date().toISOString(), 21600);
    }

    // ===== v18: アポインターの全データを取得 =====
    // クライアントがログイン時に呼び出して、最新のリスト・履歴・アポ・シフト等を取得
    if (d.type === "fetch_state") {
      if (!d.pin) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "pin required"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      var sheetName = PIN_MAP[d.pin];
      var staffName = sheetName; // 柏木・園田・堀内・河野・福田
      var fullName = FULL_NAME_MAP[d.pin] || sheetName; // 柏木幸代・園田絵里・堀内真紀子・河野由香・福田貴美恵
      // 名前マッチング関数 (どちらの名前形式でもヒット)
      var nameMatches = function(v) {
        var s = String(v||"");
        return s === staffName || s === fullName;
      };
      var ss2 = SpreadsheetApp.openById(SHEET_ID);
      var result = {
        status: "ok",
        pin: d.pin,
        name: fullName,
        timestamp: new Date().toISOString()
      };
      // 1. 架電リスト (アポインター個別シート)
      try {
        var listSh = ss2.getSheetByName(sheetName);
        if (listSh && listSh.getLastRow() > 1) {
          // v53: 5列目「元行」も返す（架電位置の復元・元シート反映の突合に使う）
          var listCols = Math.min(5, Math.max(4, listSh.getLastColumn()));
          var listData = listSh.getRange(2, 1, listSh.getLastRow() - 1, listCols).getValues();
          // 空行除外
          result.list = listData.filter(function(r){
            return r.some(function(c){return c!==null&&c!==""})
          });
        } else {
          result.list = [];
        }
      } catch (e) { result.list = []; result.listError = String(e); }

      // 2. 勤怠 (att) - 名前で絞り込み、最近30日分
      try {
        var attSh = ss2.getSheetByName("勤怠");
        result.att = {};
        if (attSh && attSh.getLastRow() > 1) {
          var attRows = attSh.getDataRange().getValues();
          for (var i = 1; i < attRows.length; i++) {
            if (nameMatches(attRows[i][0])) {
              var dateKey = formatDateKey(attRows[i][1]);
              if (!dateKey) continue;
              if (!result.att[dateKey]) result.att[dateKey] = {};

              result.att[dateKey].co = formatTimeStr(attRows[i][3]);
              result.att[dateKey].bs = formatTimeStr(attRows[i][4]);

        }
      } catch (e) { result.att = {}; result.attError = String(e); }

      // 3. 架電履歴 (hist) - 名前で絞り込み、最近1000件
      try {
        var histSh = ss2.getSheetByName("架電結果");
        result.hist = [];
        if (histSh && histSh.getLastRow() > 1) {
          var histRows = histSh.getDataRange().getValues();

            if (nameMatches(histRows[i][0])) {
              result.hist.push({
                d: formatDateKey(histRows[i][1]),
                tm: formatTimeStr(histRows[i][2]),
                co: histRows[i][3] || "",
                ph: histRows[i][4] || "",
                pe: histRows[i][5] || "",

                dur: histRows[i][7] || "",
                ck: histRows[i][8] || "",
                bs: histRows[i][9] || "",

                memo: histRows[i][11] || "",

              });
            }
          }
        }
      } catch (e) { result.hist = []; result.histError = String(e); }

      // 4. アポ (appos)

          var apoRows = apoSh.getDataRange().getValues();

            if (nameMatches(apoRows[i][1])) {  // B列=報告者
              result.appos.push({
                date: formatDateKey(apoRows[i][0]),
                reporter: apoRows[i][1] || "",
                project: apoRows[i][2] || "",

                meetingTime: formatTimeStr(apoRows[i][4]),

                personality: apoRows[i][13] || ""
              });
            }
          }
        }
      } catch (e) { result.appos = []; result.apposError = String(e); }

      // 5. 再架電 (cb)
      try {
        var cbSh = ss2.getSheetByName("再架電予定");
        result.cb = [];
        if (cbSh && cbSh.getLastRow() > 1) {
          var cbRows = cbSh.getDataRange().getValues();
          for (var i = 1; i < cbRows.length; i++) {
            if (nameMatches(cbRows[i][1])) {
              result.cb.push({
                regAt: formatDateKey(cbRows[i][0]),

        }
      } catch (e) { result.cb = []; result.cbError = String(e); }

      // v28: NGリスト (全アポインター共有) - 最新2000件
      try {
        var ngSh = ss2.getSheetByName("NGリスト");
        result.ngList = [];
        if (ngSh && ngSh.getLastRow() > 1) {
          var ngRows = ngSh.getDataRange().getValues();
          for (var i = ngRows.length - 1; i >= 1 && result.ngList.length < 2000; i--) {
            // 削除済み(状態=削除)はスキップ
            if (String(ngRows[i][5] || "") === "削除") continue;
            result.ngList.push({

      } catch (e) { result.ngList = []; result.ngError = String(e); }

      // 6. シフト (shiftDays) - 申請履歴の最新月分のみ取得
      try {
        var shftSh = ss2.getSheetByName("シフト申請");
        result.shiftHistory = [];
        if (shftSh && shftSh.getLastRow() > 1) {

            if (String(shftRows[i][3]) === d.pin || nameMatches(shftRows[i][2])) {

      } catch (e) { result.shiftHistory = []; result.shiftError = String(e); }

      // 7. v34: 面談(1on1)履歴 - 自分の記録を最新順で返す
      try {
        var mtgSh = ss2.getSheetByName("面談記録");
        result.meetingHistory = [];
        if (mtgSh && mtgSh.getLastRow() > 1) {

      try {

        if (posSh && posSh.getLastRow() > 1) {

          for (var pi = 1; pi < posRows.length; pi++) {

      return ContentService.createTextOutput(JSON.stringify(result))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ===== v32: 管理者用 全アポインターの当月シフト取得 =====
    // 「シフト申請」シートから対象月(d.month="YYYY-MM")の最新申請を全員分返す
    if (d.type === "fetch_all_shifts") {
      var targetMonth = String(d.month || "");
      var shSs = SpreadsheetApp.openById(SHEET_ID);
      var shSh = shSs.getSheetByName("シフト申請");
      var shiftResult = { status: "ok", month: targetMonth, shifts: [] };
      if (shSh && shSh.getLastRow() > 1) {
        var shRows = shSh.getDataRange().getValues();
        // 列: 0申請日時,1種別,2申請者,3PIN,4対象月,5出勤予定日数,6休み日数,7合計時間(分),8詳細,9状態
        // 申請者×対象月でグループ化し、後勝ち(appendRowは末尾追加=新しい)で最新のみ採用
        var latestMap = {};
        for (var si = 1; si < shRows.length; si++) {
          var sr = shRows[si];
          var rowMonth = String(sr[4] || "");
          if (targetMonth && rowMonth !== targetMonth) continue;
          var applicant = String(sr[2] || "");
          if (!applicant) continue;
          latestMap[applicant + "|" + rowMonth] = sr; // 上書き→最新が残る
        }
        Object.keys(latestMap).forEach(function(k){
          var sr = latestMap[k];
          shiftResult.shifts.push({
            name: String(sr[2] || ""),
            pin: String(sr[3] || ""),
            month: String(sr[4] || ""),
            workCount: Number(sr[5] || 0),
            offCount: Number(sr[6] || 0),
            totalMins: Number(sr[7] || 0),
            mode: String(sr[1] || "申請"),
            appliedAt: formatDateTimeStr(sr[0]),
            status: String(sr[9] || "受付"),
            days: parseShiftDetail(sr[8])
          });
        });
        shiftResult.shifts.sort(function(a,b){ return a.name < b.name ? -1 : (a.name > b.name ? 1 : 0); });
      }
      return ContentService.createTextOutput(JSON.stringify(shiftResult))
        .setMimeType(ContentService.MimeType.JSON);
    }

    // ===== v33: 管理者がシフトを承認/差戻し =====
    // d.pin, d.month, d.action("approve"|"reject"), d.name, d.note(任意)
    if (d.type === "shift_approve") {
      try {
        var apSs = SpreadsheetApp.openById(SHEET_ID);
        var apSh = apSs.getSheetByName("シフト申請");
        if (!apSh || apSh.getLastRow() < 2) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "error", message: "シフト申請シートが空です"
          })).setMimeType(ContentService.MimeType.JSON);
        }
        var apRows = apSh.getDataRange().getValues();
        // 末尾優先(最新)で PIN(列3) & 対象月(列4) が一致する行を探す
        var targetRow = -1;
        for (var ai = apRows.length - 1; ai >= 1; ai--) {
          if (String(apRows[ai][3]) === String(d.pin) && String(apRows[ai][4]) === String(d.month)) {
            targetRow = ai + 1; // シートは1始まり
            break;
          }
        }
        if (targetRow < 0) {
          return ContentService.createTextOutput(JSON.stringify({
            status: "error", message: "該当するシフト申請が見つかりません (" + d.pin + " / " + d.month + ")"
          })).setMimeType(ContentService.MimeType.JSON);
        }
        var newStatus = (d.action === "approve") ? "承認済" : "差戻し";
        apSh.getRange(targetRow, 10).setValue(newStatus); // 状態列(10列目)
        // 差戻し理由を詳細列(9列目)の末尾に追記
        if (d.action === "reject" && d.note) {
          var curDetail = String(apSh.getRange(targetRow, 9).getValue() || "");
          apSh.getRange(targetRow, 9).setValue(curDetail + " ／【差戻理由】" + d.note);
        }
        // 通知メール(フォールバック付き)
        try {
          var apName = d.name || FULL_NAME_MAP[d.pin] || d.pin;
          MailApp.sendEmail(
            SHIFT_NOTIFY_TO,
            "[シフト" + newStatus + "] " + apName + " " + d.month,
            apName + " さんの " + d.month + " シフトを「" + newStatus + "」にしました。" +
            (d.note ? "\n差戻理由: " + d.note : "")
          );
        } catch (apMailErr) {}
        // v45: 本人のGmailにも承認結果を通知
        try {
          var apPersonal = personalEmailFor(d.pin);
          if (apPersonal) {
            var apName2 = d.name || FULL_NAME_MAP[d.pin] || d.pin;
            MailApp.sendEmail(
              apPersonal,
              "【シフト" + newStatus + "】" + d.month + " のシフトが" + newStatus + "になりました",
              apName2 + " 様\n\n" + d.month + " のシフト申請が「" + newStatus + "」になりました。\n" +
              (d.note ? "\n■ 差戻し理由:\n" + d.note + "\n\n修正のうえ、アプリから再申請をお願いします。\n" : "\nアプリのシフト画面からもご確認いただけます。\n") +
              "\n川嶋メソッド テレアポ業務管理システム"
            );
          }
        } catch (apPerErr) {}
        logAudit("シフト" + newStatus, "管理者", (d.name || FULL_NAME_MAP[d.pin] || d.pin) + " / " + d.month, d.note || "");
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", pin: d.pin, month: d.month, newStatus: newStatus
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (apErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(apErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v33: 管理者が当月の架電実績を全アポインター分集計 (顧客請求・報酬計算用) =====
    // d.month = "YYYY-MM"。「架電結果」シートを名前×当月で集計しアポ率まで算出
    if (d.type === "fetch_call_summary") {
      try {
        var csMonth = String(d.month || "");
        var csSs = SpreadsheetApp.openById(SHEET_ID);
        var csSh = csSs.getSheetByName("架電結果");
        var csSummary = [];
        if (csSh && csSh.getLastRow() > 1) {
          var csRows = csSh.getDataRange().getValues();
          // 列: 0名前,1日付,2時刻,3会社名,4電話番号,5担当者,6結果,7通話時間,...
          var csByName = {};
          for (var ci = 1; ci < csRows.length; ci++) {
            var crow = csRows[ci];
            var cnm = String(crow[0] || "").trim();
            if (!cnm) continue;
            var cdk = normalizeYMD(crow[1]); // "yyyy-MM-dd"
            if (cdk.slice(0, 7) !== csMonth) continue; // 当月のみ
            var cres = mapResultLabelToId(crow[6]);
            var ctalk = parseDurToSec(crow[7]);
            if (!csByName[cnm]) {
              csByName[cnm] = {
                name: cnm, pin: pinForName(cnm),
                total: 0, appo: 0, cb: 0, absent: 0, refused: 0, wrong: 0, other: 0,
                talkSec: 0, days: {}
              };
            }
            var co = csByName[cnm];
            co.total++;
            if (co[cres] !== undefined && cres !== "name") { co[cres]++; } else { co.other++; }
            co.talkSec += ctalk;
            if (!co.days[cdk]) co.days[cdk] = { calls: 0, appo: 0, talkSec: 0 };
            co.days[cdk].calls++;
            if (cres === "appo") co.days[cdk].appo++;
            co.days[cdk].talkSec += ctalk;
          }
          Object.keys(csByName).forEach(function(k) {
            var co = csByName[k];
            co.reached = co.total - co.absent - co.wrong; // 到達数(不在・不通を除外)
            co.apoRate = co.reached > 0 ? Math.round(co.appo / co.reached * 1000) / 10 : 0;
            csSummary.push(co);
          });
          csSummary.sort(function(a, b) { return a.name < b.name ? -1 : (a.name > b.name ? 1 : 0); });
        }
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", month: csMonth, summary: csSummary
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (csErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(csErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v34: 面談(1on1)記録の保存 (予約/実施) - IDでupsert =====
    // d.pin, d.name, d.mid(ID), d.meetingDate, d.partner, d.cycle, d.preNote, d.postNote, d.mstatus
    if (d.type === "meeting_log") {
      try {
        var mlSs = SpreadsheetApp.openById(SHEET_ID);
        var mlSh = ensureSheet(mlSs, "面談記録",
          ["記録日時", "アポインター", "PIN", "面談日", "面談相手", "サイクル", "事前メモ", "事後メモ", "ステータス", "ID"]);
        var mlRow = [
          new Date(), d.name || "", d.pin || "", d.meetingDate || "", d.partner || "",
          d.cycle || "", d.preNote || "", d.postNote || "", d.mstatus || "予約済", d.mid || ""
        ];
        // 既存ID行を探して upsert
        var mlFound = -1;
        if (d.mid && mlSh.getLastRow() > 1) {
          var mlData = mlSh.getDataRange().getValues();
          for (var mi = mlData.length - 1; mi >= 1; mi--) {
            if (String(mlData[mi][9]) === String(d.mid)) { mlFound = mi + 1; break; }
          }
        }
        if (mlFound > 0) {
          mlSh.getRange(mlFound, 1, 1, mlRow.length).setValues([mlRow]);
        } else {
          mlSh.appendRow(mlRow);
        }
        logAudit("面談記録", d.name || "", (d.meetingDate || "") + " / " + (d.partner || ""), d.mstatus || "");
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", mid: d.mid || "", updated: mlFound > 0
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (mlErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(mlErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v34: 面談削除 (ID指定) =====
    if (d.type === "meeting_delete") {
      try {
        var mdSs = SpreadsheetApp.openById(SHEET_ID);
        var mdSh = mdSs.getSheetByName("面談記録");
        if (mdSh && mdSh.getLastRow() > 1 && d.mid) {
          var mdData = mdSh.getDataRange().getValues();
          for (var di = mdData.length - 1; di >= 1; di--) {
            if (String(mdData[di][9]) === String(d.mid)) { mdSh.deleteRow(di + 1); break; }
          }
        }
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", mid: d.mid || ""
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (mdErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(mdErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v34: 管理者(SV/課長)が全アポインターの面談状況を取得 =====
    // 面談記録を氏名でグループ化し、最新面談日・サイクル・件数・履歴を返す
    if (d.type === "fetch_meetings") {
      try {
        var fmSs = SpreadsheetApp.openById(SHEET_ID);
        var fmSh = fmSs.getSheetByName("面談記録");
        var fmByName = {};
        if (fmSh && fmSh.getLastRow() > 1) {
          var fmRows = fmSh.getDataRange().getValues();
          // 列: 0記録日時,1アポインター,2PIN,3面談日,4相手,5サイクル,6事前,7事後,8状態,9ID
          for (var fi = 1; fi < fmRows.length; fi++) {
            var fr = fmRows[fi];
            var fnm = String(fr[1] || "").trim();
            if (!fnm) continue;
            var fdate = normalizeYMD(fr[3]);
            if (!fmByName[fnm]) {
              fmByName[fnm] = { name: fnm, pin: String(fr[2] || "") || pinForName(fnm),
                lastDate: "", cycle: String(fr[5] || ""), count: 0, lastStatus: "", history: [] };
            }
            var fo = fmByName[fnm];
            fo.count++;
            fo.history.push({
              date: fdate, partner: String(fr[4] || ""), cycle: String(fr[5] || ""),
              preNote: String(fr[6] || ""), postNote: String(fr[7] || ""),
              status: String(fr[8] || ""), mid: String(fr[9] || "")
            });
            if (fdate && fdate > fo.lastDate) { fo.lastDate = fdate; fo.lastStatus = String(fr[8] || ""); fo.cycle = String(fr[5] || fo.cycle); }
          }
        }
        var fmOut = [];
        Object.keys(fmByName).forEach(function(k) {
          var fo = fmByName[k];
          fo.history.sort(function(a, b) { return a.date < b.date ? 1 : (a.date > b.date ? -1 : 0); });
          fmOut.push(fo);
        });
        fmOut.sort(function(a, b) { return a.name < b.name ? -1 : (a.name > b.name ? 1 : 0); });
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", meetings: fmOut
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (fmErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(fmErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v35: お知らせ投稿 (経営層→アポインター) - IDでupsert =====
    // d.title, d.bodyText, d.importance("normal"|"important"), d.publish(true/false), d.aid
    if (d.type === "announce_post") {
      try {
        var anSs = SpreadsheetApp.openById(SHEET_ID);
        var anSh = ensureSheet(anSs, "お知らせ",
          ["投稿日時", "タイトル", "本文", "重要度", "掲載", "ID"]);
        var anRow = [
          new Date(), d.title || "", d.bodyText || "", d.importance || "normal",
          (d.publish === false ? "false" : "true"), d.aid || ""
        ];
        var anFound = -1;
        if (d.aid && anSh.getLastRow() > 1) {
          var anData = anSh.getDataRange().getValues();
          for (var ani = anData.length - 1; ani >= 1; ani--) {
            if (String(anData[ani][5]) === String(d.aid)) { anFound = ani + 1; break; }
          }
        }
        if (anFound > 0) { anSh.getRange(anFound, 1, 1, anRow.length).setValues([anRow]); }
        else { anSh.appendRow(anRow); }
        // v45: 「重要」お知らせは登録済みの本人Gmailへ一斉送信 (アプリを開かなくても届く)
        try {
          if ((d.importance === "important") && (d.publish !== false)) {
            var anTo = broadcastEmails();
            if (anTo) {
              MailApp.sendEmail(
                anTo,
                "【重要なお知らせ】" + (d.title || ""),
                "経営層からの重要なお知らせです。\n\n■ " + (d.title || "") + "\n\n" +
                (d.bodyText || "") + "\n\n---\nアプリのホーム画面にも掲載されています。\n川嶋メソッド テレアポ業務管理システム"
              );
            }
          }
        } catch (anBcErr) {}
        logAudit(anFound > 0 ? "お知らせ更新" : "お知らせ掲載", "管理者", d.title || "", "重要度:" + (d.importance || "normal") + (d.publish === false ? " / 非掲載" : ""));
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", aid: d.aid || "", updated: anFound > 0
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (anErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(anErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v35: お知らせ削除 =====
    if (d.type === "announce_delete") {
      try {
        var adSs = SpreadsheetApp.openById(SHEET_ID);
        var adSh = adSs.getSheetByName("お知らせ");
        if (adSh && adSh.getLastRow() > 1 && d.aid) {
          var adData = adSh.getDataRange().getValues();
          for (var adi = adData.length - 1; adi >= 1; adi--) {
            if (String(adData[adi][5]) === String(d.aid)) {
              logAudit("お知らせ削除", "管理者", String(adData[adi][1] || d.aid), "");
              adSh.deleteRow(adi + 1); break;
            }
          }
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", aid: d.aid || "" }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (adErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(adErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v35: 管理者が全お知らせを取得 (掲載中/非掲載を含む) =====
    if (d.type === "fetch_announcements") {
      try {
        var faSs = SpreadsheetApp.openById(SHEET_ID);
        var faSh = faSs.getSheetByName("お知らせ");
        var faOut = [];
        if (faSh && faSh.getLastRow() > 1) {
          var faRows = faSh.getDataRange().getValues();
          for (var fai = 1; fai < faRows.length; fai++) {
            var far = faRows[fai];
            faOut.push({
              ts: formatDateTimeStr(far[0]), title: String(far[1] || ""), bodyText: String(far[2] || ""),
              importance: String(far[3] || "normal"), publish: String(far[4]) !== "false", aid: String(far[5] || "")
            });
          }
        }
        faOut.reverse(); // 新しい順
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", announcements: faOut }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (faErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(faErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v37: 監査ログ取得 (管理者) - 誰がいつ何をしたかの記録 =====
    if (d.type === "fetch_audit") {
      try {
        var auSs = SpreadsheetApp.openById(SHEET_ID);
        var auSh = auSs.getSheetByName("監査ログ");
        var auOut = [];
        if (auSh && auSh.getLastRow() > 1) {
          var auRows = auSh.getDataRange().getValues();
          var auLimit = d.limit ? Number(d.limit) : 300;
          var auStart = Math.max(1, auRows.length - auLimit);
          // 列: 0日時,1種別,2実行者,3対象,4詳細
          for (var aui = auRows.length - 1; aui >= auStart; aui--) {
            auOut.push({
              ts: formatDateTimeStr(auRows[aui][0]),
              action: String(auRows[aui][1] || ""),
              actor: String(auRows[aui][2] || ""),
              target: String(auRows[aui][3] || ""),
              detail: String(auRows[aui][4] || "")
            });
          }
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", audit: auOut }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (auErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(auErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v38: 月次勤怠(実働時間)集計 (管理者・報酬計算用) =====
    // 「退勤レポート」(ADMIN_SHEET_ID)を名前×当月で集計。同一日は重複排除(後勝ち)
    if (d.type === "fetch_attendance_summary") {
      try {
        var asMonth = String(d.month || "");
        var asSs = SpreadsheetApp.openById(ADMIN_SHEET_ID);
        var asSh = asSs.getSheetByName("退勤レポート");
        var asSummary = [];
        if (asSh && asSh.getLastRow() > 1) {
          var asRows = asSh.getDataRange().getValues();
          // 列: 0日付,1アポインター,2PIN,3出勤,4退勤,...,8実働(分),9架電数,10アポ数,...,13通話(秒)
          var asByName = {};
          for (var asi = 1; asi < asRows.length; asi++) {
            var ar = asRows[asi];
            var anm = String(ar[1] || "").trim();
            if (!anm) continue;
            var adk = normalizeYMD(ar[0]); // "yyyy-MM-dd"
            if (adk.slice(0, 7) !== asMonth) continue;
            if (!asByName[anm]) {
              asByName[anm] = { name: anm, pin: String(ar[2] || "") || pinForName(anm), days: {} };
            }
            // 同一日は後勝ち(upsert失敗時の重複行を排除)
            asByName[anm].days[adk] = {
              min: Number(ar[8]) || 0,
              calls: Number(ar[9]) || 0,
              appos: Number(ar[10]) || 0,
              talkSec: Number(ar[13]) || 0,
              ci: String(ar[3] || ""), co: String(ar[4] || "")
            };
          }
          Object.keys(asByName).forEach(function(k) {
            var o = asByName[k];
            var dkeys = Object.keys(o.days);
            var workMin = 0, calls = 0, appos = 0, talkSec = 0;
            dkeys.forEach(function(dk) {
              workMin += o.days[dk].min;
              calls += o.days[dk].calls;
              appos += o.days[dk].appos;
              talkSec += o.days[dk].talkSec;
            });
            o.workDays = dkeys.length;
            o.workMin = workMin;
            o.calls = calls;
            o.appos = appos;
            o.talkSec = talkSec;
            asSummary.push(o);
          });
          asSummary.sort(function(a, b) { return a.name < b.name ? -1 : (a.name > b.name ? 1 : 0); });
        }
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", month: asMonth, summary: asSummary
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (asErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(asErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v39: クライアント(案件)別アポ集計 (管理者・請求用) =====
    // 「アポイント」シートを登録日の当月でフィルタし、案件名×報告者で集計＋明細を返す
    if (d.type === "fetch_appo_billing") {
      try {
        var abMonth = String(d.month || "");
        var abSs = SpreadsheetApp.openById(SHEET_ID);
        var abSh = abSs.getSheetByName("アポイント");
        var abByProj = {};
        var abDetails = [];
        if (abSh && abSh.getLastRow() > 1) {
          var abRows = abSh.getDataRange().getValues();
          // 列: 0登録日,1報告者,2案件名,3商談日付,4商談時間,5法人名,6電話番号,7担当者
          for (var abi = 1; abi < abRows.length; abi++) {
            var abr = abRows[abi];
            var abdk = normalizeYMD(abr[0]); // 登録日 = 成果発生日
            if (abdk.slice(0, 7) !== abMonth) continue;
            var abProj = String(abr[2] || "").trim() || "(案件名なし)";
            var abRep = String(abr[1] || "").trim();
            if (!abByProj[abProj]) { abByProj[abProj] = { project: abProj, count: 0, byRep: {} }; }
            abByProj[abProj].count++;
            abByProj[abProj].byRep[abRep] = (abByProj[abProj].byRep[abRep] || 0) + 1;
            abDetails.push({
              date: abdk, reporter: abRep, project: abProj,
              meetingDate: normalizeYMD(abr[3]) || String(abr[3] || ""),
              meetingTime: String(abr[4] || ""),
              company: String(abr[5] || ""), phone: String(abr[6] || ""),
              person: String(abr[7] || "")
            });
          }
        }
        var abProjects = [];
        Object.keys(abByProj).forEach(function(k) { abProjects.push(abByProj[k]); });
        abProjects.sort(function(a, b) { return b.count - a.count || (a.project < b.project ? -1 : 1); });
        abDetails.sort(function(a, b) { return a.date < b.date ? -1 : (a.date > b.date ? 1 : 0); });
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", month: abMonth, projects: abProjects, details: abDetails
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (abErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(abErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v42: アカウント認証 (暗証番号) - なりすましログイン防止 =====
    // シート「アカウント認証」[PIN, 名前, ハッシュ, 設定日時]
    if (d.type === "auth_status") {
      try {
        var atSs = SpreadsheetApp.openById(SHEET_ID);
        var atSh = atSs.getSheetByName("アカウント認証");

    // ===== v46: セットアップ診断 (管理者) - 全メンバーの利用可否を一括点検 =====
    // 各PINについて: リストシート有無(無ければ自動作成)・行数・暗証番号設定・本人メール登録 を返す
    // d.sendTest=true なら登録済み本人メールへテスト送信
    if (d.type === "system_diag") {
      try {
        var dgSs = SpreadsheetApp.openById(SHEET_ID);
        var dgAuth = dgSs.getSheetByName("アカウント認証");

          try { rowCnt = mSh ? Math.max(0, mSh.getLastRow() - 1) : -1; } catch (e) { rowCnt = -1; }
          dgMembers.push({
            pin: pn,

        var dgSent = 0, dgSendErr = "";
        if (d.sendTest === true) {
          for (var dgt = 0; dgt < dgMembers.length; dgt++) {

        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(dgErr) }))

    // ===== v47: 架電リストシートの一括作成 (管理者) =====
    if (d.type === "create_list_sheets") {
      try {
        var clRes = setupMemberSheets();
        logAudit("リストシート作成", "管理者",
                 "作成:" + (clRes.created.join("・") || "なし"),
                 "既存:" + (clRes.existed.join("・") || "なし"));
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", created: clRes.created, existed: clRes.existed
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (clErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(clErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v48: 元スプレッドシート連携（設定の取得・保存・同期実行） =====
    if (d.type === "ext_config_get") {
      try {
        var ecList = readExtConfigs_();
        var ecOut = [];
        var ecPins = Object.keys(FULL_NAME_MAP);
        for (var eci = 0; eci < ecPins.length; eci++) {
          var pn = ecPins[eci];
          var found = null;
          for (var ecj = 0; ecj < ecList.length; ecj++) {
            if (String(ecList[ecj].pin) === String(pn)) { found = ecList[ecj]; break; }
          }
          ecOut.push({
            pin: pn, name: FULL_NAME_MAP[pn],
            set: !!found,
            ssId: found ? found.ssId : "", sheetName: found ? found.sheetName : "",
            gid: found ? found.gid : "",
            headerRow: found ? found.headerRow : 1,
            colCompany: found ? found.colCompany : "", colPhone: found ? found.colPhone : "",
            colPerson: found ? found.colPerson : "", colAddress: found ? found.colAddress : "",
            colName: found ? found.colName : "", colStep: found ? found.colStep : "",
            colDate: found ? found.colDate : "", colTime: found ? found.colTime : "",
            colResult: found ? found.colResult : "", colMemo: found ? found.colMemo : "",
            verify: found ? found.verify !== false : true,
            dateFmt: found ? found.dateFmt : "yyyy/MM/dd",
            overwrite: found ? found.overwrite : false,
            enabled: found ? found.enabled : true,
            lastSync: found ? found.lastSync : "", lastResult: found ? found.lastResult : ""

          .setMimeType(ContentService.MimeType.JSON);

          .setMimeType(ContentService.MimeType.JSON);

    if (d.type === "ext_config_set") {
      try {
        writeExtConfig_({
          pin: d.pin, ssId: d.ssId, sheetName: d.sheetName, headerRow: d.headerRow,
          colCompany: d.colCompany, colPhone: d.colPhone,
          colPerson: d.colPerson, colAddress: d.colAddress,
          colName: d.colName, colStep: d.colStep, colDate: d.colDate,
          colTime: d.colTime, colResult: d.colResult, colMemo: d.colMemo,

          overwrite: d.overwrite, verify: d.verify, enabled: d.enabled

          .setMimeType(ContentService.MimeType.JSON);
      } catch (esErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(esErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }

    // ===== v57: 架電位置の保存（端末内の保存が消えても続きから再開できるように） =====
    if (d.type === "pos_save") {

        var psSs = SpreadsheetApp.openById(SHEET_ID);
        var psSh = ensureSheet(psSs, "架電位置",
          ["PIN", "名前", "端末ID", "行番号", "会社名", "電話番号", "元行", "更新日時"]);
        var psRow = [String(d.pin), FULL_NAME_MAP[d.pin] || "", String(d.dev || ""),
          Number(d.i) || 0, String(d.co || ""), String(d.ph || ""), String(d.src || ""), new Date()];
        var psFound = 0;
        if (psSh.getLastRow() > 1) {

        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(psErr) }))
          .setMimeType(ContentService.MimeType.JSON);

    // ===== v58: 全メンバーの連携を一括設定（タブを名前で自動割り当て） =====
    if (d.type === "ext_bulk_setup") {
      try {
        var bsSsId = extractSsId_(d.ssId);
        var bsInfo = listSheets_(bsSsId);
        var bsSheets = bsInfo.sheets;
        // 列は代表タブ（または最初のタブ）から自動検出して全員に適用
        var bsBase = d.sheetName || (bsSheets.length ? bsSheets[0].name : "");
        var bsDet = detectColumns_(bsSsId, bsBase, d.headerRow || 1, "");
        if (!bsDet.ok) {
          return ContentService.createTextOutput(JSON.stringify(
            { status: "error", message: "列の自動検出に失敗しました: " + (bsDet.message || "") }
          )).setMimeType(ContentService.MimeType.JSON);
        }
        // メンバーごとに担当タブを決める（姓 → フルネーム → PIN の順で照合）

        var bsPins = d.pins && d.pins.length ? d.pins : Object.keys(FULL_NAME_MAP);
        for (var bi = 0; bi < bsPins.length; bi++) {
          var bp = String(bsPins[bi]);

          var hit = null;
          for (var bj = 0; bj < bsSheets.length && !hit; bj++) {
            var sn = String(bsSheets[bj].name).replace(/[\s\u3000]/g, "");
            for (var bk = 0; bk < want.length; bk++) {
              var w = String(want[bk]).replace(/[\s\u3000]/g, "");
              if (w && sn.indexOf(w) >= 0) { hit = bsSheets[bj]; break; }
            }
          }
          // 明示の割り当てが渡されていればそちらを優先
          if (d.assign && d.assign[bp]) {
            for (var bm = 0; bm < bsSheets.length; bm++) {
              if (bsSheets[bm].name === d.assign[bp]) { hit = bsSheets[bm]; break; }
            }
          }
          if (!hit) { bsMiss.push(FULL_NAME_MAP[bp] || bp); continue; }
          writeExtConfig_({
            pin: bp, ssId: bsSsId, sheetName: hit.name, gid: hit.gid,
            headerRow: d.headerRow || 1,
            colCompany: bsDet.colCompany, colPhone: bsDet.colPhone,
            colPerson: bsDet.colPerson, colAddress: bsDet.colAddress,

          });

          bsMiss.length ? "タブ未検出: " + bsMiss.join("・") : "");
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", done: bsDone, missing: bsMiss,
          sheets: bsSheets, detected: bsDet
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (bsErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(bsErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }

    // ===== v59: 全メンバーの連携を一括設定 =====

    if (d.type === "ext_bulk_setup") {

            colPerson: det.colPerson, colAddress: det.colAddress,
            colName: det.colName, colStep: det.colStep, colDate: det.colDate,
            colTime: det.colTime, colResult: det.colResult, colMemo: det.colMemo,
            dateFmt: d.dateFmt || "yyyy/MM/dd", resultMapText: "",
            overwrite: d.overwrite === true, verify: d.verify !== false, enabled: true
          });
          bsDone.push(FULL_NAME_MAP[m.pin] || m.pin);
        }
        logAudit("連携一括設定", "管理者", bsDone.length + "名を設定", bsFail.join(" / "));
        return ContentService.createTextOutput(JSON.stringify(
          { status: "ok", done: bsDone, failed: bsFail }
        )).setMimeType(ContentService.MimeType.JSON);
      } catch (bsErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(bsErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    // ===== v59: 全員分を「取り込み → 反映」まで一気に実行 =====
    if (d.type === "ext_run_all") {
      try {
        var raOut = [];
        var raCfgs = readExtConfigs_();
        for (var ri = 0; ri < raCfgs.length; ri++) {
          var c = raCfgs[ri];
          if (!c.enabled) continue;
          if (d.pin && String(c.pin) !== String(d.pin)) continue;
          var one = { pin: c.pin, name: FULL_NAME_MAP[c.pin] || c.name || "" };
          if (d.skipImport !== true) {
            var im = importFromExternal_(c);
            one.imported = im.imported || 0;
            one.importMsg = im.message || "";
            if (!im.ok) { one.ok = false; one.message = im.message; raOut.push(one); continue; }
          }
          var sy = syncExternalFor_(c, false);
          one.ok = sy.ok; one.written = sy.written || 0;
          one.unmatched = sy.unmatched || [];
          one.mismatch = sy.mismatch || [];
          one.message = (one.importMsg ? one.importMsg + " → " : "") + (sy.message || "");
          raOut.push(one);
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", reports: raOut }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (raErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(raErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== v60: 月次シフトカレンダー（管理者のみ） =====
    if (d.type === "shift_calendar") {
      try {
        var scSs = SpreadsheetApp.openById(SHEET_ID);
        var scSh = scSs.getSheetByName("シフト明細");
        var byDate = {};
        if (scSh && scSh.getLastRow() > 1) {
          var scRows = scSh.getDataRange().getValues();
          for (var sci = 1; sci < scRows.length; sci++) {
            if (String(scRows[sci][0]) !== String(d.month)) continue;
            var dk = scRows[sci][1];
            if (dk instanceof Date) { dk = Utilities.formatDate(dk, "Asia/Tokyo", "yyyy-MM-dd"); }
            dk = String(dk);
            if (!byDate[dk]) byDate[dk] = [];
            byDate[dk].push({
              pin: String(scRows[sci][2]), name: String(scRows[sci][3] || ""),
              start: String(scRows[sci][4] || ""), end: String(scRows[sci][5] || ""),
              breaks: String(scRows[sci][6] || ""), mins: Number(scRows[sci][7]) || 0
            });
          }
        }
        // 「シフトカレンダー」シートにも書き出す（スプレッドシートだけで確認できるように）
        if (d.writeSheet === true) {
          var calSh = ensureSheet(scSs, "シフトカレンダー", ["日付", "曜日", "出勤人数", "出勤者（時間）"]);
          calSh.clear();
          calSh.getRange(1, 1, 1, 4).setValues([["日付", "曜日", "出勤人数", "出勤者（時間）"]]);
          calSh.getRange(1, 1, 1, 4).setFontWeight("bold").setBackground("#e8eaf6");
          var mp = String(d.month).split("-");
          var yy = Number(mp[0]), mm = Number(mp[1]);
          var lastDay = new Date(yy, mm, 0).getDate();
          var dows = ["日", "月", "火", "水", "木", "金", "土"];
          var outRows = [];
          for (var dd = 1; dd <= lastDay; dd++) {
            var key = yy + "-" + ("0" + mm).slice(-2) + "-" + ("0" + dd).slice(-2);
            var list = byDate[key] || [];
            list.sort(function (a, b) { return a.start.localeCompare(b.start); });
            outRows.push([key, dows[new Date(yy, mm - 1, dd).getDay()], list.length,
              list.map(function (x) { return x.name + " " + x.start + "-" + x.end + (x.breaks ? "（休" + x.breaks + "）" : ""); }).join(" ／ ")]);
          }
          if (outRows.length) { calSh.getRange(2, 1, outRows.length, 4).setValues(outRows); }
          calSh.setColumnWidth(1, 100); calSh.setColumnWidth(2, 50);
          calSh.setColumnWidth(3, 80); calSh.setColumnWidth(4, 700);
          calSh.setFrozenRows(1);
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", month: d.month, byDate: byDate }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (scErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(scErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // v51: 元スプレッドシートのタブ一覧（選択肢用）
    if (d.type === "ext_sheets") {
      try {
        var lsRes = listSheets_(d.ssId);
        return ContentService.createTextOutput(JSON.stringify(
          { status: "ok", title: lsRes.title, sheets: lsRes.sheets, gidFromUrl: extractGid_(d.ssId) }
        )).setMimeType(ContentService.MimeType.JSON);
      } catch (lsErr) {
        return ContentService.createTextOutput(JSON.stringify(
          { status: "error", message: "開けません（共有設定をご確認ください）: " + lsErr }
        )).setMimeType(ContentService.MimeType.JSON);
      }
    }
    // v51: 自動反映トリガーの状態取得・設定
    if (d.type === "ext_trigger") {
      try {
        var tgRes = (d.action === "set") ? setSyncTrigger(d.on === true, d.minutes) : getSyncTriggerStatus();
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", trigger: tgRes }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (tgErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(tgErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    // v51: 設定を他メンバーへ複製（同じ元スプレッドシートでタブだけ違う運用向け）
    if (d.type === "ext_config_copy") {
      try {
        var srcList = readExtConfigs_(), src = null;
        for (var cpi = 0; cpi < srcList.length; cpi++) {
          if (String(srcList[cpi].pin) === String(d.fromPin)) { src = srcList[cpi]; break; }
        }
        if (!src) {
          return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "コピー元の設定がありません" }))
            .setMimeType(ContentService.MimeType.JSON);
        }
        var targets = d.toPins || [];
        var done = [];
        for (var cpj = 0; cpj < targets.length; cpj++) {
          var tp = String(targets[cpj]);
          if (tp === String(d.fromPin)) continue;
          writeExtConfig_({
            pin: tp, ssId: src.ssId, sheetName: PIN_MAP[tp] || "", gid: "",
            headerRow: src.headerRow,
            colCompany: src.colCompany, colPhone: src.colPhone,
            colPerson: src.colPerson, colAddress: src.colAddress,
            colName: src.colName, colStep: src.colStep, colDate: src.colDate,
            colTime: src.colTime, colResult: src.colResult, colMemo: src.colMemo,
            dateFmt: src.dateFmt, resultMapText: "",
            overwrite: src.overwrite, verify: src.verify, enabled: src.enabled
          });
          done.push(FULL_NAME_MAP[tp] || tp);
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", copied: done }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (cpErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(cpErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    // v49: 元シートの列を自動検出
    if (d.type === "ext_detect") {
      try {
        var dtRes = detectColumns_(d.ssId, d.sheetName, d.headerRow, d.gid);
        return ContentService.createTextOutput(JSON.stringify(
          { status: dtRes.ok ? "ok" : "error", detect: dtRes, message: dtRes.message || "" }
        )).setMimeType(ContentService.MimeType.JSON);
      } catch (dtErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(dtErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    // v49: 元シート → 架電リストへ取り込み
    if (d.type === "ext_import") {
      try {
        var imRep = importExternalAll(d.pin || null);
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", reports: imRep }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (imErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(imErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }
    if (d.type === "ext_sync") {
      try {
        var syRep = syncExternalAll(d.pin || null, d.dryRun === true);
        return ContentService.createTextOutput(JSON.stringify({ status: "ok", reports: syRep }))
          .setMimeType(ContentService.MimeType.JSON);
      } catch (syErr) {
        return ContentService.createTextOutput(JSON.stringify({ status: "error", message: String(syErr) }))
          .setMimeType(ContentService.MimeType.JSON);
      }
    }

    // クライアントがリスト読込時に呼び出して、GASスプレッドシートに上書き保存
    // → 他の端末・他のアカウントからもこのリストが見える
    // ===== v20: チャンク分割アップロード (大量データを安全に送信) =====
    // 1セッション内で複数のリクエストに分割して受信、最後のチャンクで完了
    if (d.type === "upload_list_chunk") {
      if (!d.pin || !d.sessionId || d.rows === undefined) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "pin, sessionId, rows required"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      var sheetName_c = PIN_MAP[d.pin];
      if (!sheetName_c) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "unknown pin"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      // セッションロック (同一PINで複数セッションが並行しないように)
      var lock_c = LockService.getScriptLock();
      var lockOk_c = false;
      try { lockOk_c = lock_c.tryLock(15000); } catch (e) {}
      if (!lockOk_c) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "lock timeout - retry later"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      try {
        var ss_c = SpreadsheetApp.openById(SHEET_ID);
        var sh_c = ss_c.getSheetByName(sheetName_c);
        if (!sh_c) {
          sh_c = ss_c.insertSheet(sheetName_c);
        }
        // 最初のチャンク → シートをクリアしてヘッダーを書く
        if (d.isFirst) {
          sh_c.clear();
          sh_c.getRange(1, 1, 1, 4).setValues([["会社名", "電話番号", "担当者", "住所"]])
            .setFontWeight("bold").setBackground("#e8eaf6");
          sh_c.setFrozenRows(1);
          // セッション情報をキャッシュに保存
          var cache_c = CacheService.getScriptCache();
          cache_c.put("upload_session_" + d.sessionId, JSON.stringify({
            pin: d.pin,
            sheetName: sheetName_c,
            chunksReceived: 0,
            totalChunks: d.totalChunks,
            startedAt: new Date().toISOString()
          }), 3600);
        }
        // データ書込み (現在の最終行の次から)
        if (d.rows && d.rows.length > 0) {
          var normalized_c = d.rows.map(function(r) {
            return [r[0] || "", r[1] || "", r[2] || "", r[3] || ""];
          });
          var startRow = sh_c.getLastRow() + 1;
          sh_c.getRange(startRow, 1, normalized_c.length, 4).setValues(normalized_c);
        }
        // セッション進捗を更新
        var cache_c2 = CacheService.getScriptCache();
        var sessKey = "upload_session_" + d.sessionId;
        var sessRaw = cache_c2.get(sessKey);
        var sess = sessRaw ? JSON.parse(sessRaw) : { chunksReceived: 0, totalChunks: d.totalChunks };
        sess.chunksReceived = (sess.chunksReceived || 0) + 1;
        cache_c2.put(sessKey, JSON.stringify(sess), 3600);

        // 最後のチャンクなら完了応答 (実際の合計件数を返す)
        if (d.isLast) {
          var totalRowsWritten = sh_c.getLastRow() - 1;  // ヘッダー除く
          Logger.log("upload_list_chunk DONE: " + sheetName_c + " - " + totalRowsWritten + " rows total");
          return ContentService.createTextOutput(JSON.stringify({
            status: "ok",
            sheet: sheetName_c,
            chunkIndex: d.chunkIndex,
            isLast: true,
            totalRows: totalRowsWritten,
            chunksReceived: sess.chunksReceived
          })).setMimeType(ContentService.MimeType.JSON);
        } else {
          return ContentService.createTextOutput(JSON.stringify({
            status: "ok",
            sheet: sheetName_c,
            chunkIndex: d.chunkIndex,
            isLast: false,
            chunksReceived: sess.chunksReceived
          })).setMimeType(ContentService.MimeType.JSON);
        }
      } finally {
        if (lockOk_c) {
          try { lock_c.releaseLock(); } catch (e) {}
        }
      }
    }

    if (d.type === "upload_list") {
      if (!d.pin || !d.rows) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "pin and rows required"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      var sheetName_u = PIN_MAP[d.pin];
      if (!sheetName_u) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: "unknown pin"
        })).setMimeType(ContentService.MimeType.JSON);
      }
      var ss_u = SpreadsheetApp.openById(SHEET_ID);
      var sh_u = ss_u.getSheetByName(sheetName_u);
      if (!sh_u) {
        sh_u = ss_u.insertSheet(sheetName_u);
      }
      // 既存内容を全削除してから書き直し
      sh_u.clear();
      // ヘッダー
      sh_u.getRange(1, 1, 1, 4).setValues([["会社名", "電話番号", "担当者", "住所"]])
        .setFontWeight("bold").setBackground("#e8eaf6");
      sh_u.setFrozenRows(1);
      // データ書込み
      if (d.rows.length > 0) {
        // 4列に揃える
        var normalized = d.rows.map(function(r) {
          return [
            r[0] || "",
            r[1] || "",
            r[2] || "",
            r[3] || ""
          ];
        });
        sh_u.getRange(2, 1, normalized.length, 4).setValues(normalized);
      }
      Logger.log("upload_list: " + sheetName_u + " - " + d.rows.length + " rows");
      return ContentService.createTextOutput(JSON.stringify({
        status: "ok",
        sheet: sheetName_u,
        rows: d.rows.length
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ===== v18: 単発で行を更新 (スキップ・架電状態の同期は別の方法で扱う) =====
    // (将来拡張用、現状は使わない)

    var ss = SpreadsheetApp.openById(SHEET_ID);

    // ===== 勤怠記録 =====
    if (d.type === "attendance") {
      var sh = ensureSheet(ss, "勤怠",
        ["名前", "日付", "出勤", "退勤", "休憩開始", "休憩終了"]);
      var rows = sh.getDataRange().getValues();
      var found = -1;
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]) === String(d.name) && String(rows[i][1]) === String(d.date)) {
          found = i + 1; break;
        }
      }
      var values = [d.name, d.date, d.clockIn, d.clockOut, d.breakStart, d.breakEnd];
      if (found > 0) {
        sh.getRange(found, 1, 1, values.length).setValues([values]);
      } else {
        sh.appendRow(values);
      }
    }

    // ===== 架電結果 =====
    if (d.type === "callResult") {
      var sh2 = ensureSheet(ss, "架電結果",
        ["名前", "日付", "時刻", "会社名", "電話番号", "担当者",
         "結果", "通話時間", "チェック数", "暴走", "切り返しアウト", "メモ", "終話ステップ", "CID", "通話秒", "元行"]);
      // v52: 同一通話の二重登録を防ぐ（直近15秒以内に同じ担当者×同じ電話番号があれば、
      //      新規追加せず最後の行を上書き＝確定した結果で置き換える）
      var dupRowNo = 0;
      try {
        var lastR = sh2.getLastRow();
        if (lastR > 1) {
          var scanFrom = Math.max(2, lastR - 20);
          var scanVals = sh2.getRange(scanFrom, 1, lastR - scanFrom + 1, 15).getValues();
          for (var dvi = scanVals.length - 1; dvi >= 0; dvi--) {
            var sv0 = scanVals[dvi];
            if (String(sv0[0] || "") !== String(d.name || "")) continue;
            if (!d.phone || String(sv0[4] || "") !== String(d.phone)) continue;
            var prevId = Number(sv0[13]) || 0;   // CID = 記録時刻(ms)
            var nowId = Number(d.cid) || 0;
            if (prevId && nowId && Math.abs(nowId - prevId) <= 15000) { dupRowNo = scanFrom + dvi; }
            break;
          }
        }
      } catch (dupErr) {}
      var crRow = [
        d.name, d.date, d.time,
        d.company, d.phone, d.person,
        d.result, d.duration, d.checkedCount,
        d.bousou, d.outText, d.memo,
        d.endStep || "", d.cid || "", Number(d.talkSec) || 0,
        Number(d.srcRow) || ""
      ];
      if (dupRowNo > 0) { sh2.getRange(dupRowNo, 1, 1, crRow.length).setValues([crRow]); }
      else { sh2.appendRow(crRow); }
      // v43: 文字起こし本文は専用シートへ (CIDで架電結果と紐づけ)
      if (d.transcript) {
        try {
          var trSh = ensureSheet(ss, "文字起こし",
            ["日時", "名前", "会社名", "電話番号", "結果", "通話時間", "CID", "本文"]);
          trSh.appendRow([
            new Date(), d.name || "", d.company || "", d.phone || "",
            d.result || "", d.duration || "", d.cid || "",
            String(d.transcript).slice(0, 45000)
          ]);
        } catch (trErr) {}
      }
    }

    // ===== v36: 架電結果の取消 (誤タップ復旧) - CIDで直近行を削除 =====
    if (d.type === "callResult_undo") {
      try {
        var cuSs = SpreadsheetApp.openById(SHEET_ID);
        var cuSh = cuSs.getSheetByName("架電結果");
        var cuDeleted = false;
        if (cuSh && cuSh.getLastRow() > 1 && d.cid) {
          var cuData = cuSh.getDataRange().getValues();
          // CIDは末尾列(index 13)。名前も一致する直近行を末尾優先で削除
          for (var cui = cuData.length - 1; cui >= 1; cui--) {
            if (String(cuData[cui][13]) === String(d.cid) &&
                (!d.name || String(cuData[cui][0]) === String(d.name))) {
              logAudit("架電結果取消", d.name || String(cuData[cui][0] || ""), String(cuData[cui][3] || ""), "結果:" + String(cuData[cui][6] || ""));
              cuSh.deleteRow(cui + 1);
              cuDeleted = true;
              break;
            }
          }
        }
        // v43: 紐づく文字起こしも削除
        try {
          var cuTr = cuSs.getSheetByName("文字起こし");
          if (cuTr && cuTr.getLastRow() > 1 && d.cid) {
            var cuTrData = cuTr.getDataRange().getValues();
            for (var cuti = cuTrData.length - 1; cuti >= 1; cuti--) {
              if (String(cuTrData[cuti][6]) === String(d.cid)) { cuTr.deleteRow(cuti + 1); break; }
            }
          }
        } catch (cuTrErr) {}
        return ContentService.createTextOutput(JSON.stringify({
          status: "ok", cid: d.cid || "", deleted: cuDeleted
        })).setMimeType(ContentService.MimeType.JSON);
      } catch (cuErr) {
        return ContentService.createTextOutput(JSON.stringify({
          status: "error", message: String(cuErr)
        })).setMimeType(ContentService.MimeType.JSON);
      }
    }

    // ===== 再架電予定 =====
    if (d.type === "callback") {
      var sh3 = ensureSheet(ss, "再架電予定",
        ["登録日時", "名前", "再架電予定日時", "会社名", "電話番号", "担当者", "メモ", "状態"]);
      sh3.appendRow([
        new Date(), d.name, d.scheduledAt,
        d.company, d.phone, d.person, d.memo, "未対応"
      ]);
    }

    // ===== アポイント詳細 (v11: 商談日付/時間 追加) =====
    if (d.type === "appointment") {
      var sh4 = ensureSheet(ss, "アポイント",
        ["登録日", "報告者", "案件名", "商談日付", "商談時間",
         "法人名", "電話番号", "担当者", "性別", "役職",
         "メール", "住所", "アポ日時", "人柄",
         "特定技能案件", "外国人雇用", "特定技能", "技人国", "面接・採用決定者", "録音送付"]);
      sh4.appendRow([
        d.date, d.reporter, d.project, d.meetingDate || "", d.meetingTime || "",
        d.company, d.phone, d.person, d.gender, d.role,
        d.email, d.address, d.datetime, d.personality,
        d.isTokutei || "", d.gaikokujin || "", d.tokutei || "",
        d.gijinkoku || "", d.decider || "", d.recSent || ""
      ]);
    }

    // ===== シフト申請 (v12: 時間範囲対応) =====
    if (d.type === "shift_apply" || d.type === "shift_modify") {
      var isModify = d.type === "shift_modify";
      var sh5 = ensureSheet(ss, "シフト申請",
        ["申請日時", "種別", "申請者", "PIN", "対象月",
         "出勤予定日数", "休み日数", "合計時間(分)", "詳細", "状態"]);
      var detail = "";
      if (d.days) {
        var keys = Object.keys(d.days).sort();
        detail = keys.map(function(k){
          var v = d.days[k];
          if (v === "off") return k + ": 休み";
          if (v && typeof v === "object" && v.start && v.end) {
            return k + ": " + v.start + "〜" + v.end;
          }
          return k + ": " + String(v);
        }).join(" / ");
      }
      // v60: 日別明細を保存（管理者の月次カレンダー用）。同じ人・同じ月は入れ替える
      try {
        var dtSh = ensureSheet(ss, "シフト明細",
          ["対象月", "日付", "PIN", "名前", "開始", "終了", "休憩", "実働(分)", "更新日時"]);
        if (dtSh.getLastRow() > 1) {
          var dtAll = dtSh.getDataRange().getValues();
          for (var dti = dtAll.length - 1; dti >= 1; dti--) {
            if (String(dtAll[dti][0]) === String(d.month) && String(dtAll[dti][2]) === String(d.pin)) {
              dtSh.deleteRow(dti + 1);
            }
          }
        }
        var dtRows = [];
        if (d.days) {
          Object.keys(d.days).sort().forEach(function (k) {
            var v = d.days[k];
            if (!v || v === "off") return;
            if (typeof v !== "object" || !v.start || !v.end) return;
            var brTxt = (v.breaks || []).map(function (b) { return b.start + "-" + b.end; }).join(" / ");
            var wm = 0;
            try {
              var toMin = function (t) { return parseInt(String(t).slice(0, 2)) * 60 + parseInt(String(t).slice(3, 5)); };
              wm = Math.max(0, toMin(v.end) - toMin(v.start));
              (v.breaks || []).forEach(function (b) {
                if (!b || !b.start || !b.end) return;
                var bs = Math.max(toMin(b.start), toMin(v.start));
                var be = Math.min(toMin(b.end), toMin(v.end));
                if (be > bs) wm -= (be - bs);
              });
              wm = Math.max(0, wm);
            } catch (e) {}
            dtRows.push([d.month, k, d.pin, d.name, v.start, v.end, brTxt, wm, new Date()]);
          });
        }
        if (dtRows.length) { dtSh.getRange(dtSh.getLastRow() + 1, 1, dtRows.length, 9).setValues(dtRows); }
      } catch (dtErr) { Logger.log("shift detail save failed: " + dtErr); }

      sh5.appendRow([
        new Date(),
        isModify ? "修正" : "申請",
        d.name, d.pin, d.month,
        d.workCount || 0,
        d.offCount || 0,
        d.totalMins || 0,
        detail,
        isModify ? "修正受付" : "受付"
      ]);
      // メール送信 (v15: フォールバック付き)
      try {
        var to = d.to || SHIFT_NOTIFY_TO;
        var subject = d.subject || ((isModify ? "【シフト修正申請】" : "【シフト申請】") + d.month + " " + d.name);
        var body = d.body || ("申請者: " + d.name + "\n対象月: " + d.month + "\n\n" + detail);
        try {
          MailApp.sendEmail({to: to, subject: subject, body: body});
          Logger.log("Shift mail sent via MailApp to " + to);
        } catch (e1) {
          try {
            GmailApp.sendEmail(to, subject, body);
            Logger.log("Shift mail sent via GmailApp to " + to);
          } catch (e2) {
            Logger.log("Shift mail failed (both): " + e1 + " | " + e2);
          }
        }
      } catch (mailErr) {
        Logger.log("Shift mail wrapper error: " + mailErr);
      }
    }

    // ===== 退勤レポート (v14追加, v17: LockService追加) =====
    // 管理者用スプレッドシートに書込＋メール通知
    if (d.type === "daily_report") {
      // v17: 4人同時退勤時のシート競合を防ぐため LockService で排他制御

      }
    }

    return ContentService.createTextOutput(JSON.stringify({status:"ok", type: d.type}))

  } catch (err) {

// v18: 日付・時刻フォーマット用ヘルパー
function formatDateKey(v) {
  if (!v) return "";
  if (v instanceof Date) {
    return Utilities.formatDate(v, "Asia/Tokyo", "yyyy-MM-dd");
  }
  var s = String(v);
  // "2026/5/4" → "2026-05-04"
  var m = s.match(/^(\d{4})[\/-](\d{1,2})[\/-](\d{1,2})/);
  if (m) {
    return m[1] + "-" + ("0"+m[2]).slice(-2) + "-" + ("0"+m[3]).slice(-2);
  }
  return s;
}
function formatTimeStr(v) {
  if (!v) return "";
  if (v instanceof Date) {
    return Utilities.formatDate(v, "Asia/Tokyo", "HH:mm:ss");
  }
  var s = String(v);
  // "9:00" → "09:00:00", "9:00:00" → そのまま
  var m = s.match(/^(\d{1,2}):(\d{1,2})(?::(\d{1,2}))?/);
  if (m) {
    return ("0"+m[1]).slice(-2) + ":" + ("0"+m[2]).slice(-2) + ":" + (m[3]?("0"+m[3]).slice(-2):"00");
  }
  return s;
}
function mapResultLabelToId(label) {
  if (!label) return "";
  var s = String(label).trim();
  // 絵文字付きラベルにも対応
  if (s.indexOf("アポ") >= 0) return "appo";
  if (s.indexOf("再架電") >= 0) return "cb";
  if (s.indexOf("不在") >= 0) return "absent";
  if (s.indexOf("断り") >= 0 || s.indexOf("拒否") >= 0) return "refused";
  if (s.indexOf("不通") >= 0) return "wrong";
  if (s.indexOf("その他") >= 0) return "other";
  // すでにIDだった場合はそのまま
  return s;
}

    // 上限管理: 5500行を超えたら古い1000行をまとめて削除 (稀にしか走らない)
    var auLast = auSh.getLastRow();
    if (auLast > 5500) { auSh.deleteRows(2, 1000); }
  } catch (e) {}
}

function formatDateTimeStr(v) {
  if (!v) return "";
  try {
    var d = (v instanceof Date) ? v : new Date(v);
    if (isNaN(d.getTime())) return String(v);
    return Utilities.formatDate(d, "Asia/Tokyo", "yyyy-MM-dd HH:mm");
  } catch (e) { return String(v); }
}

// v32: 「シフト申請」シートの詳細文字列を日別オブジェクトに復元
//   "2026-06-01: 09:30〜18:00 / 2026-06-02: 休み / 2026-06-03: 13:00〜16:45"
//   → { "2026-06-01": {start:"09:30",end:"18:00"}, "2026-06-02": "off", ... }
function parseShiftDetail(detail) {
  var days = {};
  if (!detail) return days;
  String(detail).split(" / ").forEach(function(part) {
    var idx = part.indexOf(": ");
    if (idx < 0) return;
    var key = part.slice(0, idx).trim();
    var val = part.slice(idx + 2).trim();
    if (!/^\d{4}-\d{2}-\d{2}$/.test(key)) return;
    if (val === "休み" || val === "off") { days[key] = "off"; return; }
    var pr = val.split("〜");
    if (pr.length === 2) days[key] = { start: pr[0].trim(), end: pr[1].trim() };
  });
  return days;
}

// v33: 日付を "yyyy-MM-dd" に正規化 (Date / "2026/6/1" / "2026年6月1日" 対応)
function normalizeYMD(v) {
  if (!v) return "";
  if (v instanceof Date) {
    return Utilities.formatDate(v, "Asia/Tokyo", "yyyy-MM-dd");
  }
  var s = String(v);
  var m = s.match(/^(\d{4})[\/\-年](\d{1,2})[\/\-月](\d{1,2})/);
  if (m) {
    return m[1] + "-" + ("0" + m[2]).slice(-2) + "-" + ("0" + m[3]).slice(-2);
  }
  return s;
}

// v33: 通話時間文字列 "M:SS" / "H:MM:SS" を秒に変換
function parseDurToSec(v) {
  if (!v) return 0;
  if (typeof v === "number") return Math.round(v);
  var parts = String(v).trim().split(":");
  if (parts.length === 2) {
    return (parseInt(parts[0], 10) || 0) * 60 + (parseInt(parts[1], 10) || 0);
  }
  if (parts.length === 3) {
    return (parseInt(parts[0], 10) || 0) * 3600 + (parseInt(parts[1], 10) || 0) * 60 + (parseInt(parts[2], 10) || 0);
  }
  return 0;
}

// v33: 全名 → PIN (FULL_NAME_MAP の逆引き / 見つからなければ空)
function pinForName(name) {
  var n = String(name || "").trim();
  var keys = Object.keys(FULL_NAME_MAP);
  for (var i = 0; i < keys.length; i++) {
    if (FULL_NAME_MAP[keys[i]] === n) return keys[i];
  }
  // 短縮名(PIN_MAP)でも照合
  var k2 = Object.keys(PIN_MAP);
  for (var j = 0; j < k2.length; j++) {
    if (PIN_MAP[k2[j]] === n) return k2[j];
  }
  return "";
}

function ensureSheet(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    if (headers && headers.length) {
      sh.appendRow(headers);
      sh.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e8eaf6");
      sh.setFrozenRows(1);
    }
  } else if (headers && headers.length) {
    // v15: 既存シートのヘッダーが古い場合、不足列を追加
    var lastCol = sh.getLastColumn();
    if (lastCol < headers.length) {
      var existingHeaders = lastCol > 0 ? sh.getRange(1, 1, 1, lastCol).getValues()[0] : [];
      var missingHeaders = headers.slice(lastCol);
      sh.getRange(1, lastCol + 1, 1, missingHeaders.length).setValues([missingHeaders])
        .setFontWeight("bold").setBackground("#e8eaf6");
    }
  }
  return sh;
}

// 会社名の正規化（表記ゆらぎを吸収してズレを防ぐ）
function normCompany_(s) {
  if (s === null || s === undefined) return "";
  var t = String(s);
  t = t.replace(/[Ａ-Ｚａ-ｚ０-９]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); });
  t = t.replace(/[\s\u3000]/g, "");
  t = t.replace(/[㈱]/g, "株式会社").replace(/[（(]株[）)]/g, "株式会社");
  t = t.replace(/[㈲]/g, "有限会社").replace(/[（(]有[）)]/g, "有限会社");
  t = t.replace(/[（(]社[）)]/g, "社団法人").replace(/[（(]財[）)]/g, "財団法人");
  t = t.replace(/[（(]医[）)]/g, "医療法人").replace(/[（(]福[）)]/g, "社会福祉法人");
  t = t.replace(/[‐‑‒–—―ー−\-]/g, "ー");
  t = t.replace(/[・･,，.．'’"”`｀]/g, "");
  t = t.replace(/[（）()【】\[\]「」『』]/g, "");
  return t.toLowerCase();
}
// 法人格を除いた核名称（前株/後株のゆらぎ用・補助照合）
function coreCompany_(s) {
  return normCompany_(s).replace(
    /(株式会社|有限会社|合同会社|合資会社|合名会社|一般社団法人|公益社団法人|社団法人|一般財団法人|財団法人|医療法人社団|医療法人|社会福祉法人|学校法人|特定非営利活動法人|npo法人)/g, "");
}
// 電話番号は数字のみ（最も確実なキー）
function normPhone_(s) {
  if (!s) return "";
  var t = String(s).replace(/[０-９]/g, function (c) { return String.fromCharCode(c.charCodeAt(0) - 0xFEE0); });
  return t.replace(/[^0-9]/g, "");
}
// 列指定 "A"/"AB"/3 → 1始まりの列番号（0=未設定）
function colToIndex_(col) {
  if (col === null || col === undefined || col === "") return 0;
  if (typeof col === "number") return col;
  var s = String(col).trim().toUpperCase();
  if (/^[0-9]+$/.test(s)) return parseInt(s, 10);
  var n = 0;
  for (var i = 0; i < s.length; i++) {
    var c = s.charCodeAt(i) - 64;
    if (c < 1 || c > 26) return 0;
    n = n * 26 + c;
  }
  return n;
}

}
// "アポ=アポ獲得,不在=不在" 形式の変換表
function parseResultMap_(s) {
  var map = {};
  String(s || "").split(/[,、\n]/).forEach(function (pair) {
    var kv = pair.split("=");
    if (kv.length === 2) {
      var k = kv[0].trim(), v = kv[1].trim();
      if (k) map[k] = v;
    }
  });
  return map;
}

// ===== v49: 元スプレッドシートの列を見出しから自動検出 =====
function detectColumns_(ssId, sheetName, headerRow, gid) {
  var out = { ok: false };
  try {
    var ss0 = SpreadsheetApp.openById(extractSsId_(ssId));
    var sh = resolveSheet_(ss0, sheetName, gid || extractGid_(ssId));   // v51: gidでも特定
    if (!sh) { out.message = "シートが見つかりません（シート名またはGIDをご確認ください）"; return out; }
    out.sheetName = sh.getName();
    out.gid = String(sh.getSheetId());
    var hr = Number(headerRow) || 1;
    var lastCol = sh.getLastColumn();
    if (lastCol < 1) { out.message = "列がありません"; return out; }
    var hdr = sh.getRange(hr, 1, 1, lastCol).getValues()[0];
    var norm = hdr.map(function (h) { return String(h || "").replace(/[\s\u3000]/g, "").toLowerCase(); });
    function find(pats, exclude) {
      for (var i = 0; i < norm.length; i++) {
        if (!norm[i]) continue;
        if (exclude) {
          var skip = false;
          for (var e = 0; e < exclude.length; e++) { if (norm[i].indexOf(exclude[e]) >= 0) { skip = true; break; } }
          if (skip) continue;
        }
        for (var j = 0; j < pats.length; j++) { if (norm[i].indexOf(pats[j]) >= 0) return i + 1; }
      }
      return 0;
    }
    out.ok = true;
    out.header = hdr.map(function (h) { return String(h || ""); });
    out.colCompany = find(["会社", "法人", "施設", "企業", "名称", "事業所"]);
    out.colPhone = find(["電話", "tel", "番号"], ["法人番号", "整理番号"]);
    out.colPerson = find(["担当", "氏名", "施設長", "代表者"]);
    out.colAddress = find(["住所", "所在地"]);
    out.colDate = find(["日付", "架電日", "実施日"]);
    out.colTime = find(["時刻", "時間"], ["通話時間"]);
    out.colName = find(["名前", "担当者名", "アポインター", "架電者"], ["会社", "法人", "施設"]);
    out.colResult = find(["結果", "ステータス", "状況"]);
    out.colStep = find(["終話ステップ", "ステップ", "どこまで"]);
    out.colMemo = find(["メモ", "備考", "特記", "コメント"]);
    return out;
  } catch (e) {
    out.message = String(e);
    return out;
  }
}

// ===== v51: 自動反映トリガーの設置・解除（GASエディタを開かずにアプリから操作） =====
var AUTO_SYNC_FN = "autoSyncExternal";
function getSyncTriggerStatus() {
  var trs = ScriptApp.getProjectTriggers();
  for (var i = 0; i < trs.length; i++) {
    if (trs[i].getHandlerFunction() === AUTO_SYNC_FN) {
      return { on: true, id: trs[i].getUniqueId() };
    }
  }
  return { on: false };
}
function setSyncTrigger(on, minutes) {
  var trs = ScriptApp.getProjectTriggers();
  for (var i = 0; i < trs.length; i++) {
    if (trs[i].getHandlerFunction() === AUTO_SYNC_FN) { ScriptApp.deleteTrigger(trs[i]); }
  }
  if (!on) { logAudit("自動反映", "管理者", "停止", ""); return { on: false }; }
  var m = Number(minutes) || 60;
  var b = ScriptApp.newTrigger(AUTO_SYNC_FN).timeBased();
  if (m >= 60) { b.everyHours(Math.max(1, Math.round(m / 60))).create(); }
  else { b.everyMinutes([1, 5, 10, 15, 30].indexOf(m) >= 0 ? m : 15).create(); }
  logAudit("自動反映", "管理者", "開始（" + m + "分ごと）", "");

}

// 1名分を同期する本体

  var results = [];
  for (var k = 1; k < crRows.length; k++) {
    var rr = crRows[k];
    if (String(rr[0] || "").trim() !== String(myName).trim()) continue;
    results.push({
      date: normalizeYMD(rr[1]), time: String(rr[2] || ""), name: String(rr[0] || ""),
      company: String(rr[3] || ""), phone: String(rr[4] || ""), result: String(rr[6] || ""),
      memo: String(rr[11] || ""), step: String(rr[12] || ""),   // v50: メモ・終話ステップ
      srcRow: Number(rr[15]) || 0,   // v49: 元スプレッドシートの行番号
      seq: k
    });
  }
  if (!results.length) { rep.message = "この方の架電結果がありません"; return rep; }
  results.sort(function (a, b) {
    var d = (a.date + " " + a.time).localeCompare(b.date + " " + b.time);
    return d !== 0 ? d : (a.seq - b.seq);
  });

  // キーごとにグループ化（電話番号優先 → 会社名 → 核名称）
  var groups = {};
  for (var m = 0; m < results.length; m++) {
    var r = results[m];
    var key = null, rows = null;
    // v49: 元行が記録されていれば、その行に確定（会社名照合を経由しない＝ズレゼロ）
    if (r.srcRow && r.srcRow >= startRow && r.srcRow <= lastRow) {
      key = "R:" + r.srcRow; rows = [r.srcRow];
      if (!groups[key]) groups[key] = { rows: rows, res: [] };
      groups[key].res.push(r);
      continue;
    }
    var ph = normPhone_(r.phone);
    if (ph && byPhone[ph]) { key = "P:" + ph; rows = byPhone[ph]; }
    if (!rows) {
      var cn = normCompany_(r.company);
      if (cn && byComp[cn]) { key = "C:" + cn; rows = byComp[cn]; }
    }
    if (!rows) {
      var co = coreCompany_(r.company);
      // 核名称は「双方で一意」のときだけ採用（誤マッチ防止）
      if (co && byCore[co] && byCore[co].length === 1) { key = "K:" + co; rows = byCore[co]; }
    }
    if (!rows) { rep.unmatched.push(r.company + "（" + r.date + "）"); continue; }
    if (!groups[key]) groups[key] = { rows: rows, res: [] };
    groups[key].res.push(r);
  }

  // 行への割り当て（元行が足りない場合は最後の行に集約＝最新が残る）
  var assign = {};  // rowNo -> result
  Object.keys(groups).forEach(function (key) {
    var g = groups[key];
    for (var i = 0; i < g.res.length; i++) {
      var rowNo = (i < g.rows.length) ? g.rows[i] : g.rows[g.rows.length - 1];
      assign[rowNo] = g.res[i];  // 後勝ち＝最新の結果が残る
    }
  });

  // v50: 書き込み直前の二重チェック — 行の会社名が結果の会社名と一致するか検証
  // （取り込み後に元シートを並べ替え/行挿入した場合の誤爆を防ぐ）
  var mismatch = [];
  if (cfg.verify !== false) {
    var okAssign = {};
    for (var rk in assign) {
      var ridx = Number(rk) - startRow;
      var sheetCo = (ridx >= 0 && ridx < nRows) ? normCompany_(compVals[ridx][0]) : "";
      var resCo = normCompany_(assign[rk].company);
      if (sheetCo && resCo && sheetCo !== resCo && coreCompany_(compVals[ridx][0]) !== coreCompany_(assign[rk].company)) {
        mismatch.push(assign[rk].company + " → " + String(compVals[ridx][0]) + "（" + rk + "行目）");
        continue;   // 一致しない行には書かない
      }
      okAssign[rk] = assign[rk];
    }
    assign = okAssign;
  }
  rep.mismatch = mismatch;

  // v50: 確認モードでは、どの行に何が入るかの明細を返す
  if (dryRun) {
    rep.ok = true;
    rep.written = Object.keys(assign).length;
    rep.preview = [];
    var pvKeys = Object.keys(assign).sort(function (a, b) { return Number(a) - Number(b); });
    for (var pv = 0; pv < Math.min(10, pvKeys.length); pv++) {
      var pk = pvKeys[pv], pr = assign[pk];
      var pidx = Number(pk) - startRow;
      rep.preview.push({
        row: Number(pk),
        sheetCompany: (pidx >= 0 && pidx < nRows) ? String(compVals[pidx][0]) : "",
        name: pr.name, step: pr.step, date: pr.date, time: pr.time,
        result: cfg.resultMap[pr.result] || pr.result,
        memo: String(pr.memo || "").slice(0, 40)
      });
    }
    rep.message = "確認のみ（書き込みなし）／反映予定 " + rep.written + "件"
      + (rep.unmatched.length ? " ・未マッチ " + rep.unmatched.length + "件" : "")
      + (mismatch.length ? " ・会社名不一致 " + mismatch.length + "件" : "");
    return rep;
  }

  // 列ごとにまとめて書き込む（既存の他列は触らない）
  // v50: 名前／終話ステップ／日付／時刻／結果／メモ の6項目を書き戻す
  var targets = [
    { col: cfg.colName, get: function (r) { return r.name; } },
    { col: cfg.colStep, get: function (r) { return r.step; } },
    { col: cfg.colDate, get: function (r) { return r.date ? Utilities.formatDate(new Date(r.date + "T00:00:00+09:00"), "Asia/Tokyo", cfg.dateFmt) : ""; } },
    { col: cfg.colTime, get: function (r) { return r.time; } },
    { col: cfg.colResult, get: function (r) { return cfg.resultMap[r.result] || r.result; } },
    { col: cfg.colMemo, get: function (r) { return String(r.memo || "").slice(0, 500); } }
  ];
  var wrote = 0, skipped = 0;
  for (var t = 0; t < targets.length; t++) {
    var tg = targets[t];
    if (!tg.col) continue;
    var colVals = tsh.getRange(startRow, tg.col, nRows, 1).getValues();
    var changed = false;
    for (var rowNo in assign) {
      var idx = Number(rowNo) - startRow;
      if (idx < 0 || idx >= nRows) continue;
      var cur = colVals[idx][0];
      var val = tg.get(assign[rowNo]);
      if (val === "" || val === null) continue;
      if (!cfg.overwrite && String(cur || "").trim() !== "" && String(cur).trim() !== String(val)) {
        skipped++; continue;  // 既に手入力がある場合は尊重
      }
      if (String(cur) !== String(val)) { colVals[idx][0] = val; changed = true; }
    }
    if (changed) { tsh.getRange(startRow, tg.col, nRows, 1).setValues(colVals); }
  }
  wrote = Object.keys(assign).length;

  rep.ok = true; rep.written = wrote; rep.skipped = skipped;
  rep.message = "反映 " + wrote + "件 / 未マッチ " + rep.unmatched.length + "件"
    + (mismatch.length ? " / 会社名不一致(書込回避) " + mismatch.length + "件" : "")
    + (skipped ? " / 既存値を尊重 " + skipped + "件" : "");

  // 設定シートに結果を記録
  try {
    var csh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(EXT_CONFIG_SHEET);
    if (csh && cfg.row) { csh.getRange(cfg.row, 22, 1, 2).setValues([[new Date(), rep.message]]); }
  } catch (e) {}
  logAudit("元シート反映", "管理者", rep.name, rep.message);
  return rep;
}

  for (var i = 0; i < pins.length; i++) {
    var nm = PIN_MAP[pins[i]];
    var sh = ss.getSheetByName(nm);
    if (sh) {
      // 既存だがヘッダーが無い場合のみ補う (データは触らない)

  return { created: created, existed: existed };
}

// v16: 未送信メールの手動再送
// 「未送信メール」シートに記録されたものを翌日以降に再送する
// 必要に応じてトリガー (毎日 9時実行) で自動化推奨
// ==========================================
function resendUnsent() {
  try {
    var adminSs = SpreadsheetApp.openById(ADMIN_SHEET_ID);
    var sh = adminSs.getSheetByName("未送信メール");
    if (!sh) {
      Logger.log("未送信メールシートが存在しません - 全て送信済み");
      return "no unsent";
    }
    var data = sh.getDataRange().getValues();
    if (data.length <= 1) {
      Logger.log("未送信なし");
      return "empty";
    }
    var quota = -1;
    try { quota = MailApp.getRemainingDailyQuota(); } catch(e) {}
    if (quota === 0) {
      Logger.log("クォータ枯渇 - 翌日以降に再実行してください");
      return "quota exhausted";
    }
    var sent = 0, failed = 0;
    var rowsToDelete = [];
    for (var i = 1; i < data.length; i++) {
      if (quota > 0 && sent >= quota - 5) {
        Logger.log("クォータ温存のため停止 (送信:" + sent + ")");
        break;
      }
      try {
        MailApp.sendEmail({to: data[i][1], subject: data[i][2], body: data[i][3]});
        sent++;
        rowsToDelete.push(i + 1);
      } catch (e) {
        failed++;
        Logger.log("再送失敗 row " + (i+1) + ": " + e);
      }
    }
    // 送信済みの行を削除 (下から削除して行番号のずれを回避)
    rowsToDelete.sort(function(a,b){return b-a});
    rowsToDelete.forEach(function(r){sh.deleteRow(r)});
    var msg = "送信:" + sent + " / 失敗:" + failed;
    Logger.log("再送完了: " + msg);
    return msg;
  } catch (e) {
    Logger.log("resendUnsent error: " + e);
    return "error: " + e;
  }
}

// v16: 健全性チェック - 毎時実行のトリガー推奨
// SQに溜まっている件数を上限超えしていたら管理者に警告
// ==========================================
function healthCheck() {
  try {
    var quota = MailApp.getRemainingDailyQuota();
    Logger.log("Daily mail quota remaining: " + quota);
    if (quota < 5) {
      // 緊急通知 (1日1通限定にしたいので、最後の通知時刻をPropertiesで管理)
      var props = PropertiesService.getScriptProperties();
      var lastWarn = props.getProperty("lastQuotaWarn");
      var today = Utilities.formatDate(new Date(), "Asia/Tokyo", "yyyy-MM-dd");
      if (lastWarn !== today) {
        try {
          MailApp.sendEmail({
            to: DAILY_REPORT_TO,
            subject: "【警告】メール送信クォータ残少 - テレアポ管理",
            body: "本日のメール送信クォータが残り " + quota + " 通です。\n" +
                  "退勤通知が届かない可能性があります。"
          });
          props.setProperty("lastQuotaWarn", today);
        } catch (e) {}
      }
    }
    return "ok - quota:" + quota;
  } catch (e) {
    Logger.log("healthCheck error: " + e);
    return "error: " + e;
  }
}

// アポ獲得録音の転送先
var APO_AUDIO_FORWARD_TO = "a.kusunoki@sukimagroup.jp";

// メール添付の最大サイズ (25MB - Gmailの制限は25MB)
var MAIL_ATTACHMENT_MAX_BYTES = 24 * 1024 * 1024;

// 1回の実行で処理する最大件数 (タイムアウト対策)
var MAX_PROCESS_PER_RUN = 20;

// 通話時刻と録音ファイルのマッチング許容時間範囲 (ミリ秒)
var TIME_MATCH_TOLERANCE_MS = 60 * 60 * 1000;  // 60分

// ==========================================================================
// メイン関数: アポ獲得通話の録音を自動転送
// 設定: トリガー → 時間主導型 → 5分おき実行
// ==========================================================================
function forwardApoRecordings() {
  var startTime = new Date();
  Logger.log("=== forwardApoRecordings 開始: " + startTime.toISOString() + " ===");

  try {
    var ss = SpreadsheetApp.openById(SHEET_ID);
    var adminSs = SpreadsheetApp.openById(ADMIN_SHEET_ID);

    // 処理済みシートを準備
    var processedSh = ensureSheet(adminSs, "処理済みアポ録音", [
      "処理日時", "アポインター", "日付", "時刻", "電話番号", "会社名",
      "ファイル名", "ファイルID", "状態", "Driveフォルダ"
    ]);

    // 既処理キーを取得 (電話番号|日付|時刻|アポインター)
    var processedRows = processedSh.getDataRange().getValues();
    var processedKeys = {};
    for (var i = 1; i < processedRows.length; i++) {
      var k = String(processedRows[i][1] || "") + "|" +     // アポインター
              String(processedRows[i][2] || "") + "|" +     // 日付
              String(processedRows[i][3] || "") + "|" +     // 時刻
              String(processedRows[i][4] || "");             // 電話番号
      processedKeys[k] = true;
    }
    Logger.log("既処理レコード数: " + Object.keys(processedKeys).length);

    // 架電結果シート読み込み
    var callSh = ss.getSheetByName("架電結果");
    if (!callSh || callSh.getLastRow() < 2) {
      Logger.log("架電結果シートにデータなし");
      return { ok: true, processed: 0, message: "no calls" };
    }
    var callRows = callSh.getDataRange().getValues();

    // 本日と前日のみ対象 (パフォーマンス考慮)
    var today = new Date();
    var yesterday = new Date(today.getTime() - 24 * 60 * 60 * 1000);
    var todayKey = Utilities.formatDate(today, "Asia/Tokyo", "yyyy/MM/dd");
    var yesterdayKey = Utilities.formatDate(yesterday, "Asia/Tokyo", "yyyy/MM/dd");

    // 各アポインターのDriveフォルダIDを事前取得 (一度だけ)
    var folderCache = getAppointerFolders();
    Logger.log("Driveフォルダ取得数: " + Object.keys(folderCache).length);

    var processed = 0, skipped = 0, errors = 0, notFound = 0;

    // 新しい順に処理 (最新のアポを優先)
    for (var i = callRows.length - 1; i >= 1; i--) {
      if (processed >= MAX_PROCESS_PER_RUN) {
        Logger.log("MAX_PROCESS_PER_RUN到達 (" + MAX_PROCESS_PER_RUN + ")");
        break;
      }

      var row = callRows[i];
      var apoName = String(row[0] || "");
      var dateStr = String(row[1] || "");
      var timeStr = String(row[2] || "");
      var company = String(row[3] || "");
      var phone = String(row[4] || "");
      var person = String(row[5] || "");
      var result = String(row[6] || "");

      // フィルタ1: アポ獲得のみ
      if (result.indexOf("アポ") < 0) continue;

      // フィルタ2: 本日・前日のみ
      if (dateStr !== todayKey && dateStr !== yesterdayKey) continue;

      // 重複チェック
      var key = apoName + "|" + dateStr + "|" + timeStr + "|" + phone;
      if (processedKeys[key]) {
        skipped++;
        continue;
      }

      // アポインター姓を抽出
      var lastName = extractLastName(apoName);
      if (!lastName || !APPOINTER_DRIVE_MAP[lastName]) {
        Logger.log("アポインター不明: " + apoName);
        errors++;
        continue;
      }

      var driveInfo = APPOINTER_DRIVE_MAP[lastName];
      var folder = folderCache[lastName];
      if (!folder) {
        Logger.log("Driveフォルダ未取得: " + driveInfo.folder_name +
                   " - 管理者と" + driveInfo.email + "の共有を確認");
        // 24h以上前なら諦める (ファイル未発見として記録)
        var callTimeMs = parseCallDateTime(dateStr, timeStr);
        if ((Date.now() - callTimeMs) > 24 * 60 * 60 * 1000) {
          processedSh.appendRow([
            new Date(), apoName, dateStr, timeStr, phone, company,
            "(フォルダ未共有)", "", "未処理", driveInfo.folder_name
          ]);
        }
        errors++;
        continue;
      }

      // Drive内で録音ファイル検索
      var matchedFile = findMatchingRecording(folder, phone, dateStr, timeStr);

      if (!matchedFile) {
        // 24h以上前なら諦めて記録
        var callTime = parseCallDateTime(dateStr, timeStr);
        if ((Date.now() - callTime) > 24 * 60 * 60 * 1000) {
          processedSh.appendRow([
            new Date(), apoName, dateStr, timeStr, phone, company,
            "(録音ファイル未発見)", "", "未発見", driveInfo.folder_name
          ]);
          notFound++;
        }
        continue;
      }

      // メール送信
      var sendResult = sendRecordingMail(matchedFile, apoName, dateStr, timeStr, company, phone, person);
      if (sendResult.ok) {
        processedSh.appendRow([
          new Date(), apoName, dateStr, timeStr, phone, company,
          matchedFile.name, matchedFile.id, sendResult.status, driveInfo.folder_name
        ]);
        processed++;
        Logger.log("✅ 転送完了: " + apoName + " " + dateStr + " " + timeStr +
                   " " + company + " (" + sendResult.status + ")");
      } else {
        Logger.log("❌ 送信失敗: " + sendResult.error);
        errors++;
      }
    }

    var endTime = new Date();
    var elapsed = (endTime - startTime) / 1000;
    Logger.log("=== 完了: 処理 " + processed + "件, スキップ " + skipped +
               "件, 未発見 " + notFound + "件, エラー " + errors + "件, 経過 " +
               elapsed.toFixed(1) + "秒 ===");

    return {
      ok: true,
      processed: processed,
      skipped: skipped,
      notFound: notFound,
      errors: errors,
      elapsedSec: elapsed
    };

  } catch (e) {
    Logger.log("致命的エラー: " + e + "\n" + (e.stack || ""));
    return { ok: false, error: String(e) };
  }
}

// ==========================================================================
// アポインターのDriveフォルダを一括取得 (フォルダ名で検索)
// 各アポインターは自分のマイドライブに「{姓}_CallRecordings」を作成し、
// 管理者アカウントに編集者として共有しておく必要がある
// ==========================================================================
function getAppointerFolders() {
  var cache = {};
  for (var lastName in APPOINTER_DRIVE_MAP) {
    var folderName = APPOINTER_DRIVE_MAP[lastName].folder_name;
    try {
      var folders = DriveApp.getFoldersByName(folderName);
      if (folders.hasNext()) {
        var folder = folders.next();
        cache[lastName] = folder;
      } else {
        Logger.log("⚠️ フォルダ未発見: " + folderName +
                   " (アポインターから管理者へ共有が必要)");
      }
    } catch (e) {
      Logger.log("フォルダ検索エラー (" + folderName + "): " + e);
    }
  }
  return cache;
}

// ==========================================================================
// 録音ファイル検索 (電話番号 + 時刻でマッチング)
// Galaxy ファイル名: "通話記録 [電話番号] [yymmdd]_[HHmmss].m4a"
// 例: "通話記録 0312345678 250505_143000.m4a"
//
// 戻り値: {id, name, file, sizeBytes} または null
// ==========================================================================
function findMatchingRecording(folder, phone, dateStr, timeStr) {
  // 電話番号を数字のみに正規化
  var phoneNumeric = String(phone).replace(/[^0-9]/g, "");
  if (!phoneNumeric || phoneNumeric.length < 6) return null;

  // 検索キー: 電話番号の先頭6桁以上
  var searchPrefix = phoneNumeric.slice(0, 7);  // 7桁以上で検索精度UP
  if (searchPrefix.length < 6) searchPrefix = phoneNumeric.slice(0, 6);

  // 通話日時 (ミリ秒)
  var callTimeMs = parseCallDateTime(dateStr, timeStr);

  var bestMatch = null;
  var bestDiff = Infinity;

  try {
    var files = folder.getFiles();
    while (files.hasNext()) {
      var f = files.next();
      var fname = f.getName();

      // 電話番号マッチ
      var fnameNumeric = fname.replace(/[^0-9]/g, "");
      if (fnameNumeric.indexOf(searchPrefix) < 0) continue;

      // 音声ファイルのみ (.m4a, .mp3, .wav, .amr)
      var isAudio = /\.(m4a|mp3|wav|amr|aac)$/i.test(fname);
      if (!isAudio) {
        var mt = f.getMimeType();
        if (!mt || mt.indexOf("audio") < 0) continue;
      }

      // 時刻マッチ (許容範囲内で最も近いもの)
      var fileTimeMs = f.getLastUpdated().getTime();
      var diff = Math.abs(fileTimeMs - callTimeMs);

      if (diff <= TIME_MATCH_TOLERANCE_MS && diff < bestDiff) {
        bestMatch = f;
        bestDiff = diff;
      }
    }
  } catch (e) {
    Logger.log("ファイル検索エラー: " + e);
    return null;
  }

  if (!bestMatch) return null;

  return {
    id: bestMatch.getId(),
    name: bestMatch.getName(),
    file: bestMatch,
    sizeBytes: bestMatch.getSize(),
    diffMin: Math.floor(bestDiff / 60000)
  };
}

// ==========================================================================
// メール送信 (添付 or リンク自動切替)
// ==========================================================================
function sendRecordingMail(matchedFile, apoName, dateStr, timeStr, company, phone, person) {
  var subject = "【アポ録音】" + apoName + " " + dateStr + " " + timeStr +
                " - " + company;

  var bodyBase = "アポ獲得通話の録音ファイルです。\n\n" +
                 "■ アポインター: " + apoName + "\n" +
                 "■ 日付: " + dateStr + "\n" +
                 "■ 時刻: " + timeStr + "\n" +
                 "■ 会社名: " + company + "\n" +
                 "■ 担当者: " + (person || "未記入") + "\n" +
                 "■ 電話番号: " + phone + "\n" +
                 "■ ファイル名: " + matchedFile.name + "\n" +
                 "■ ファイルサイズ: " + (matchedFile.sizeBytes / 1024 / 1024).toFixed(2) + " MB\n" +
                 "■ 時刻マッチ誤差: " + matchedFile.diffMin + "分\n\n";

  // サイズチェック - 25MB超ならリンク送信
  if (matchedFile.sizeBytes > MAIL_ATTACHMENT_MAX_BYTES) {
    try {
      var fileUrl = matchedFile.file.getUrl();
      MailApp.sendEmail({
        to: APO_AUDIO_FORWARD_TO,
        subject: subject + " (リンク送信)",
        body: bodyBase + "▼ ファイルサイズが大きいため、Driveリンクで共有します:\n" +
              fileUrl + "\n\n" +
              "----- 川嶋メソッド アポ録音転送システム -----\n",
        name: "川嶋メソッド アポ録音転送"
      });
      return { ok: true, status: "リンク送信" };
    } catch (e) {
      return { ok: false, error: "リンク送信失敗: " + e };
    }
  }

  // 通常添付送信
  try {
    var blob = matchedFile.file.getBlob();
    MailApp.sendEmail({
      to: APO_AUDIO_FORWARD_TO,
      subject: subject,
      body: bodyBase + "▼ 録音ファイルを添付しています。\n\n" +
            "----- 川嶋メソッド アポ録音転送システム -----\n",
      attachments: [blob],
      name: "川嶋メソッド アポ録音転送"
    });
    return { ok: true, status: "添付送信" };
  } catch (e) {
    // 添付エラー → リンク送信フォールバック
    Logger.log("添付エラー、リンク送信に切替: " + e);
    try {
      var fallbackUrl = matchedFile.file.getUrl();
      MailApp.sendEmail({
        to: APO_AUDIO_FORWARD_TO,
        subject: subject + " (リンク送信)",
        body: bodyBase + "▼ 添付エラーのため、Driveリンクで共有:\n" + fallbackUrl + "\n",
        name: "川嶋メソッド アポ録音転送"
      });
      return { ok: true, status: "リンク送信(フォールバック)" };
    } catch (e2) {
      return { ok: false, error: String(e2) };
    }
  }
}

// ==========================================================================
// アポインター名から姓を抽出
// "河野由香" → "河野", "柏木幸代" → "柏木"
// ==========================================================================
function extractLastName(fullName) {
  if (!fullName) return null;
  for (var lname in APPOINTER_DRIVE_MAP) {
    if (fullName.indexOf(lname) === 0) return lname;  // 先頭一致のみ
  }
  return null;
}

// ==========================================================================
// 日付・時刻文字列をミリ秒に変換
// "2026/05/05" + "14:30" → タイムスタンプ
// ==========================================================================
function parseCallDateTime(dateStr, timeStr) {
  try {
    var d = String(dateStr || "").replace(/[年月]/g, "/").replace(/日/g, "");
    var parts = d.split(/[\/\-]/);
    if (parts.length < 3) return Date.now();
    var y = parseInt(parts[0], 10);
    var m = parseInt(parts[1], 10) - 1;
    var dy = parseInt(parts[2], 10);
    var t = String(timeStr || "00:00").split(":");
    var h = parseInt(t[0], 10) || 0;
    var mn = parseInt(t[1], 10) || 0;
    return new Date(y, m, dy, h, mn, 0).getTime();
  } catch (e) {
    return Date.now();
  }
}
